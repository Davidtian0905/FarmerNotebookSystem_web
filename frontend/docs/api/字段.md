```java
// 基于系统当前时间作为基准
public class TimeDimensionCalculator {
    
    // 今日：当天00:00:00到23:59:59
    public static LocalDate getToday() {
        return LocalDate.now();
    }
    
    // 昨日：昨天00:00:00到23:59:59
    public static LocalDate getYesterday() {
        return LocalDate.now().minusDays(1);
    }
    
    // 本周：从周一到周日（或从周日到周六，根据业务需求）
    public static LocalDateRange getThisWeek() {
        LocalDate today = LocalDate.now();
        LocalDate monday = today.with(TemporalAdjusters.previousOrSame(DayOfWeek.MONDAY));
        LocalDate sunday = today.with(TemporalAdjusters.nextOrSame(DayOfWeek.SUNDAY));
        return new LocalDateRange(monday, sunday);
    }
    
    // 本月：从1号到月末
    public static LocalDateRange getThisMonth() {
        LocalDate today = LocalDate.now();
        LocalDate firstDay = today.withDayOfMonth(1);
        LocalDate lastDay = today.with(TemporalAdjusters.lastDayOfMonth());
        return new LocalDateRange(firstDay, lastDay);
    }
    
    // 本年：从1月1日到12月31日
    public static LocalDateRange getThisYear() {
        LocalDate today = LocalDate.now();
        LocalDate firstDay = today.withDayOfYear(1);
        LocalDate lastDay = today.withDayOfYear(today.lengthOfYear());
        return new LocalDateRange(firstDay, lastDay);
    }
}
```

### 1. 时间维度计算服务
```java
@Service
public class FinancialCalculationService {
    
    // 计算指定日期的财务数据
    public DailyFinancialSummary calculateDailySummary(Long userId, LocalDate date) {
        // 收入计算
        BigDecimal dailyIncome = transactionMapper.sumIncomeByDate(userId, date);
        int incomeCount = transactionMapper.countIncomeByDate(userId, date);
        
        // 支出计算
        BigDecimal dailyExpense = transactionMapper.sumExpenseByDate(userId, date);
        int expenseCount = transactionMapper.countExpenseByDate(userId, date);
        
        // 利润计算
        BigDecimal dailyProfit = dailyIncome.subtract(dailyExpense);
        
        return new DailyFinancialSummary(dailyIncome, dailyExpense, dailyProfit, incomeCount, expenseCount);
    }
    
    // 计算指定周期的财务数据
    public PeriodFinancialSummary calculatePeriodSummary(Long userId, LocalDateRange dateRange) {
        // 收入计算
        BigDecimal periodIncome = transactionMapper.sumIncomeByDateRange(userId, dateRange);
        int incomeCount = transactionMapper.countIncomeByDateRange(userId, dateRange);
        
        // 支出计算
        BigDecimal periodExpense = transactionMapper.sumExpenseByDateRange(userId, dateRange);
        int expenseCount = transactionMapper.countExpenseByDateRange(userId, dateRange);
        
        // 利润计算
        BigDecimal periodProfit = periodIncome.subtract(periodExpense);
        
        return new PeriodFinancialSummary(periodIncome, periodExpense, periodProfit, incomeCount, expenseCount);
    }
    
    // 计算收入分类统计
    public List<IncomeCategorySummary> calculateIncomeCategorySummary(Long userId, LocalDateRange dateRange) {
        return transactionMapper.sumIncomeByCategory(userId, dateRange);
    }
}
```

### 2. 图表数据生成服务
```java
@Service
public class ChartDataService {
    
    // 生成收入趋势图表数据
    public ChartData generateIncomeTrendChart(Long userId, String period, LocalDateRange dateRange) {
        List<DailyFinancialData> dailyData = new ArrayList<>();
        
        LocalDate currentDate = dateRange.getStartDate();
        while (!currentDate.isAfter(dateRange.getEndDate())) {
            DailyFinancialSummary summary = calculateDailySummary(userId, currentDate);
            dailyData.add(new DailyFinancialData(currentDate, summary));
            currentDate = currentDate.plusDays(1);
        }
        
        return new ChartData(period, dailyData);
    }
    
    // 生成收入分类饼图数据
    public PieChartData generateIncomeCategoryPieChart(Long userId, LocalDateRange dateRange) {
        List<IncomeCategorySummary> categories = calculateIncomeCategorySummary(userId, dateRange);
        
        BigDecimal totalIncome = categories.stream()
            .map(IncomeCategorySummary::getAmount)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        // 计算百分比
        categories.forEach(category -> {
            BigDecimal percentage = category.getAmount()
                .divide(totalIncome, 4, RoundingMode.HALF_UP)
                .multiply(new BigDecimal("100"));
            category.setPercentage(percentage);
        });
        
        return new PieChartData(totalIncome, categories);
    }
}
```
